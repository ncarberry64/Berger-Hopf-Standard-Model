The median integral in section 3 can also be evaluated directly:

The following standalone code block requires Python and SymPy. It starts

All matrices, source and moving constraint are specified. The scalar t is

physical coupling. Variables are ordered `(q1,q2,z1,z2,lambda)`.

```python
import sympy as s
x, t = s.symbols("x t", real=True)
W = s.sin(x)**2 * s.cos(x)**2
dW = s.sin(4*x) * s.sin(2*x) / 2
Z = s.integrate(W, (x, 0, s.pi/2))
dZ = s.integrate(dW, (x, 0, s.pi/2))
dA = s.integrate(dW, (x, 0, s.pi/4))
shift = s.simplify((dZ/2-dA)/W.subs(x, s.pi/4))
assert (Z, dZ, dA, shift) == (s.pi/16, 0, s.Rational(1,6), -s.Rational(2,3))
print("median:", Z, dZ, dA, shift)

P = s.Matrix([[5+t, 1+t*t], [1+t*t, 6-t]])
B = s.Matrix([[1, t], [2+t, 1]])
L = s.Matrix([[4+t, 1], [1, 3+t*t]])
C = s.Matrix([[1+t, 1-t]])
J = s.Matrix([1+t*t, 2-t])
d = s.Matrix([1+t])
M = (P.row_join(B).row_join(C.T)
     .col_join(B.T.row_join(L).row_join(s.zeros(2,1)))
     .col_join(C.row_join(s.zeros(1,2)).row_join(s.zeros(1,1))))
f = (-J).col_join(s.zeros(2,1)).col_join(d)
z = M.inv()*f
z0 = z.subs(t,0)
z1 = s.diff(z,t).subs(t,0)
z2 = s.diff(z,t,2).subs(t,0)
M0, M1, M2 = [s.diff(M,t,k).subs(t,0) for k in range(3)]
j1 = M0.inv()*(s.diff(f,t).subs(t,0)-M1*z0)
j2 = M0.inv()*(s.diff(f,t,2).subs(t,0)-M2*z0-2*M1*j1)
r1 = (z1-j1).applyfunc(s.simplify)
r2 = (z2-j2).applyfunc(s.simplify)
assert r1 == r2 == s.zeros(5,1)
assert L.subs(t,0)*s.diff(L,t).subs(t,0) != s.diff(L,t).subs(t,0)*L.subs(t,0)
E = P-B*L.inv()*B.T
K = E.row_join(C.T).col_join(C.row_join(s.zeros(1,1)))
y = K.inv()*((-J).col_join(d))
r0 = (y.subs(t,0)-s.Matrix([z0[0],z0[1],z0[4]])).applyfunc(s.simplify)
rs = (s.diff(y,t).subs(t,0)-s.Matrix([z1[0],z1[1],z1[4]])).applyfunc(s.simplify)
```
